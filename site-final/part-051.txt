  }
}

function showMessage(text, type = "success") {
  if (!els.success) return;
  els.success.textContent = text;
  els.success.style.display = "block";
  els.success.style.background =
    type === "error" ? "rgba(122,38,48,.16)" : "rgba(31,111,56,.14)";
  els.success.style.borderColor =
    type === "error" ? "rgba(190,70,82,.35)" : "rgba(58,160,86,.35)";
  els.success.style.color = type === "error" ? "#f1bec4" : "#c8f2d2";
}

async function loadApprovedTestimonials() {
  if (!isConfigured) {
    els.setupNotice.style.display = "block";
    els.cards.innerHTML =
      '<div class="testimonial-empty">Os depoimentos aprovados aparecerão aqui após a configuração do Firebase.</div>';
    els.approvedCount.textContent = "Aguardando configuração";
    return;
  }

  try {
    const q = query(
      collection(db, "testimonials"),
      where("status", "==", "approved"),
      orderBy("createdAt", "desc"),
      limit(4)
    );
    const snap = await getDocs(q);
    const items = snap.docs.map(d => ({ id: d.id, ...d.data() }));

    els.approvedCount.textContent =
      `${items.length} ${items.length === 1 ? "publicado" : "publicados"}`;

    if (!items.length) {
      els.cards.innerHTML =
        '<div class="testimonial-empty">Os primeiros depoimentos aprovados aparecerão aqui.</div>';
      return;
    }

    els.cards.innerHTML = items.map(item => {
      const author = escapeHtml(item.name || "Aluno(a)");
      const className = item.className ? ` · ${escapeHtml(item.className)}` : "";
      return `
        <article class="testimonial-card">
          <p>“${escapeHtml(item.testimonial)}”</p>
          <footer>
            <span class="student-name">${author}${className}</span>
            <span>${formatDate(item.createdAt)}</span>
          </footer>
        </article>
      `;
    }).join("");
  } catch (error) {
    console.error(error);
    els.cards.innerHTML =
      '<div class="testimonial-empty">Não foi possível carregar os depoimentos agora.</div>';
    els.approvedCount.textContent = "Indisponível";
  }
}
