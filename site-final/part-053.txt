function safeCanvaUrl(raw, allowEmbed = false) {
  try {
    const u = new URL(raw);
    const validHost = /(^|\.)canva\.com$/i.test(u.hostname) || u.hostname === "canva.link";
    if (u.protocol !== "https:" || !validHost) return "";
    if (u.hostname !== "canva.link" && !u.pathname.includes("/design/")) return "";
    if (allowEmbed && u.hostname !== "canva.link") { u.pathname = u.pathname.replace(/\/edit\/?$/, "/view"); u.search = "?embed"; }
    return u.toString();
  } catch { return ""; }
}
async function compressCover(file) {
  if (!file || !/^image\/(png|jpeg|webp)$/.test(file.type)) throw new Error("Escolha uma imagem JPG, PNG ou WebP.");
  if (file.size > 12 * 1024 * 1024) throw new Error("A imagem original deve ter no máximo 12 MB.");
  const bitmap = await createImageBitmap(file);
  const scale = Math.min(1, 1100 / bitmap.width, 750 / bitmap.height);
  const canvas = document.createElement("canvas");
  canvas.width = Math.max(1, Math.round(bitmap.width * scale));
  canvas.height = Math.max(1, Math.round(bitmap.height * scale));
  canvas.getContext("2d").drawImage(bitmap, 0, 0, canvas.width, canvas.height);
  bitmap.close?.();
  let quality = 0.78, data = canvas.toDataURL("image/jpeg", quality);
  while (data.length > 520000 && quality > 0.42) { quality -= 0.08; data = canvas.toDataURL("image/jpeg", quality); }
  if (data.length > 650000) throw new Error("A imagem ficou muito grande. Escolha uma imagem mais simples ou menor.");
  return data;
}

function renderLessons(items) {
  const gradeKeys = ["6ano", "9ano", "1medio", "2medio", "3medio"];
  const labels = {"6ano":"6º ano", "9ano":"9º ano", "1medio":"1º ano", "2medio":"2º ano", "3medio":"3º ano"};
  const grids = Object.fromEntries(gradeKeys.map(key => [key, els.lessonGrid?.querySelector(`[data-grade-grid="${key}"]`)]));
  gradeKeys.forEach(key => { if (grids[key]) grids[key].innerHTML = ""; });
  const grouped = Object.fromEntries(gradeKeys.map(key => [key, []]));
  (items || []).forEach(item => { if (item.published === false) return; const key = gradeKeys.includes(item.grade) ? item.grade : "6ano"; grouped[key].push(item); });
  gradeKeys.forEach(key => {
    if (!grids[key]) return;
    if (!grouped[key].length) {
      grids[key].innerHTML = `<div class="lesson-empty">As aulas do ${labels[key]} aparecerão aqui em breve.</div>`;
      return;
    }
    grids[key].innerHTML = grouped[key].map(item => {
      const title = escapeHtml(item.title || "Aula");
      const description = escapeHtml(item.description || "Apresentação de Ciências.");
      const url = safeCanvaUrl(item.canvaUrl || item.url || "");
      const cover = item.coverData ? `<img class="lesson-cover" src="${item.coverData}" alt="Capa: ${title}" loading="lazy">` : `<div class="lesson-cover lesson-cover-placeholder"><span>✦</span><small>APRESENTAÇÃO</small></div>`;
      if (!url) return "";
      const embedUrl = safeCanvaUrl(item.embedUrl || url, true);
      const preview = embedUrl ? `<div class="lesson-preview" style="position:relative;width:100%;aspect-ratio:16/9;background:#f3f4f6;overflow:hidden"><iframe src="${embedUrl}" title="Prévia da aula: ${title}" loading="eager" allow="fullscreen" allowfullscreen style="position:absolute;inset:0;width:100%;height:100%;border:0"></iframe></div>` : cover;
      return `<article class="lesson-card">${preview}<div class="lesson-content"><span class="lesson-subject">${escapeHtml(item.subject || "Aula")}</span><h3>${title}</h3><p>${description}</p><div class="lesson-actions"><a href="${url}" target="_blank" rel="noopener noreferrer">Abrir apresentação ↗</a></div></div></article>`;
    }).join("") || '<div class="lesson-empty">Nenhum link válido disponível nesta turma.</div>';
  });
}

// Os links curtos do Canva abrem em nova aba. A prévia incorporada só é usada
// quando o Canva fornece um endereço /design/... compatível.
function canvaUrlIsAllowed(raw) {
  try {
    const u = new URL(raw);
    return u.protocol === "https:" && (/(^|\.)canva\.com$/i.test(u.hostname) || u.hostname === "canva.link");
  } catch { return false; }
}

async function loadLessons() {
  let items = [...starterLessons];
  if (isConfigured && db) {
    try {
      const snap = await getDocs(query(collection(db, "lessons"), orderBy("createdAt", "desc"), limit(100)));
      const remote = snap.docs.map(d => ({id:d.id, ...d.data()})).filter(x => x.published !== false);
      // Aulas cadastradas pelo professor aparecem antes das aulas iniciais.
      items = [...remote, ...starterLessons];
    } catch (error) { console.warn("Não foi possível carregar aulas adicionais do Firestore.", error); }
  }
  renderLessons(items);
}
