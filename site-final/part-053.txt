  const item = btn.closest(".admin-item");
  const id = item?.dataset.id;
  const action = btn.dataset.action;
  if (!id) return;

  try {
    btn.disabled = true;

    if (action === "delete") {
      if (!confirm("Excluir este depoimento definitivamente?")) return;
      await deleteDoc(doc(db, "testimonials", id));
    } else {
      await updateDoc(doc(db, "testimonials", id), {
        status: action,
        moderatedAt: serverTimestamp(),
        moderatedBy: auth.currentUser.uid
      });
    }

    await Promise.all([
      loadAdminTestimonials(auth.currentUser),
      loadApprovedTestimonials()
    ]);
  } catch (error) {
    console.error(error);
    alert("Não foi possível concluir a ação. Confira as regras do Firestore.");
  } finally {
    btn.disabled = false;
  }
});

function safeCanvaUrl(raw, allowEmbed = false) {
  try {
    const u = new URL(raw);
    if (u.protocol !== "https:" || !/(^|\.)canva\.com$/i.test(u.hostname)) return "";
    if (!u.pathname.includes("/design/")) return "";
    if (allowEmbed && !u.searchParams.has("embed")) u.searchParams.set("embed", "");
    return u.toString();
  } catch { return ""; }
}

async function compressCover(file) {
  if (!file || !/^image\/(png|jpeg|webp)$/.test(file.type)) throw new Error("Escolha uma imagem JPG, PNG ou WebP.");
  if (file.size > 12 * 1024 * 1024) throw new Error("A imagem original deve ter no máximo 12 MB.");
  const bitmap = await createImageBitmap(file);
  const scale = Math.min(1, 1100 / bitmap.width, 750 / bitmap.height);
  const canvas = document.createElement("canvas");
  canvas.width = Math.max(1, Math.round(bitmap.width * scale));
  canvas.height = Math.max(1, Math.round(bitmap.height * scale));
  canvas.getContext("2d").drawImage(bitmap, 0, 0, canvas.width, canvas.height);
  bitmap.close?.();
  let quality = 0.78, data = canvas.toDataURL("image/jpeg", quality);
  while (data.length > 520000 && quality > 0.42) { quality -= 0.08; data = canvas.toDataURL("image/jpeg", quality); }
  if (data.length > 650000) throw new Error("A imagem ficou muito grande. Escolha uma imagem mais simples ou menor.");
  return data;
}
