      `${items.length} ${items.length === 1 ? "publicado" : "publicados"}`;

    if (!items.length) {
      els.cards.innerHTML =
        '<div class="testimonial-empty">Os primeiros depoimentos aprovados aparecerão aqui.</div>';
      return;
    }

    els.cards.innerHTML = items.map(item => {
      const author = escapeHtml(item.name || "Aluno(a)");
      const className = item.className ? ` · ${escapeHtml(item.className)}` : "";
      return `
        <article class="testimonial-card">
          <p>“${escapeHtml(item.testimonial)}”</p>
          <footer>
            <span class="student-name">${author}${className}</span>
            <span>${formatDate(item.createdAt)}</span>
          </footer>
        </article>
      `;
    }).join("");
  } catch (error) {
    console.error(error);
    els.cards.innerHTML =
      '<div class="testimonial-empty">Não foi possível carregar os depoimentos agora.</div>';
    els.approvedCount.textContent = "Indisponível";
  }
}

els.form?.addEventListener("submit", async (e) => {
  e.preventDefault();

  if (!isConfigured) {
    els.setupNotice.style.display = "block";
    showMessage("O envio será ativado assim que o professor concluir a configuração do Firebase.", "error");
    return;
  }

  if (els.honeypot.value.trim()) return; // provável bot

  const testimonial = els.text.value.trim();
  if (!testimonial) return;

  const lastSubmit = Number(localStorage.getItem("rafael_testimonial_last_submit") || 0);
  if (Date.now() - lastSubmit < 60_000) {
    showMessage("Aguarde um minuto antes de enviar outro depoimento.", "error");
    return;
  }

  els.submit.disabled = true;
  els.submit.textContent = "Enviando...";

  try {
    await addDoc(collection(db, "testimonials"), {
      name: els.name.value.trim().slice(0, 80),
      className: els.className.value.trim().slice(0, 40),
      testimonial: testimonial.slice(0, 700),
      status: "pending",
      createdAt: serverTimestamp()
    });

    localStorage.setItem("rafael_testimonial_last_submit", String(Date.now()));
    els.form.reset();
    showMessage("Obrigado. Seu depoimento foi enviado para análise e não ficará público automaticamente.");
  } catch (error) {
    console.error(error);
    showMessage("Não foi possível enviar agora. Tente novamente em alguns instantes.", "error");
  } finally {
    els.submit.disabled = false;
    els.submit.textContent = "Enviar depoimento";
  }
});
